-- AimLock.lua | Luau | Roblox (executor) | camera aimbot only, no HUD

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera

-- ============ CONFIG ============
local CONFIG = {
    aim_part      = "Head",       -- "Head" | "HumanoidRootPart" | "UpperTorso"
    fov           = 250,          -- pixels radius*2; target must be within fov/2 of crosshair
    smoothness    = 0.35,         -- 0..1; higher = snappier
    team_check    = true,
    dead_check    = true,
    max_distance  = 1000,         -- studs
    toggle_key    = Enum.KeyCode.Q,
    lock_enabled  = true,
}
-- =================================

local target = nil

local function get_aim_part(char)
    if not char then return nil end
    return char:FindFirstChild(CONFIG.aim_part)
        or char:FindFirstChild("Head")
        or char:FindFirstChild("HumanoidRootPart")
end

local function is_alive(player)
    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

local function screen_distance(world_pos)
    local sp, on_screen = Camera:WorldToViewportPoint(world_pos)
    if not on_screen then return math.huge end
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    return (Vector2.new(sp.X, sp.Y) - center).Magnitude
end

local function find_target()
    local best, best_dist = nil, math.huge
    local lc = LocalPlayer.Character
    local lpos = lc and lc:FindFirstChild("HumanoidRootPart") and lc.HumanoidRootPart.Position

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        if CONFIG.team_check and LocalPlayer.Team and plr.Team == LocalPlayer.Team then continue end
        if CONFIG.dead_check and not is_alive(plr) then continue end

        local part = get_aim_part(plr.Character)
        if not part then continue end

        if lpos then
            if (part.Position - lpos).Magnitude > CONFIG.max_distance then continue end
        end

        local sd = screen_distance(part.Position)
        if sd > CONFIG.fov / 2 then continue end
        if sd < best_dist then best, best_dist = plr, sd end
    end
    return best
end

RunService.RenderStepped:Connect(function()
    if not CONFIG.lock_enabled then target = nil return end

    if target and (not target.Parent or not is_alive(target)) then target = nil end
    if not target then target = find_target() end

    if target then
        local part = get_aim_part(target.Character)
        if part and screen_distance(part.Position) <= CONFIG.fov / 2 then
            local goal = CFrame.new(Camera.CFrame.Position, part.Position)
            Camera.CFrame = Camera.CFrame:Lerp(goal, math.clamp(CONFIG.smoothness, 0, 1))
        else
            target = nil
        end
    end
end)

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == CONFIG.toggle_key then
        CONFIG.lock_enabled = not CONFIG.lock_enabled
        if not CONFIG.lock_enabled then target = nil end
    end
end)
