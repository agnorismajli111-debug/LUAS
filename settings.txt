-- ------------------------------------------------------------
-- KEYBIND WINDOW
-- ------------------------------------------------------------
local _keybindWindow = nil
local _keybindRows = {}

local function _optIsOn(flag)
    local opt = Options[flag] or Toggles[flag]
    if not opt then return false end
    local v = opt.Value
    if type(v) == "boolean" then return v end
    if type(v) == "table" then return v.Value == true end
    return false
end

local _keyMap = {
    AimbotEnabled       = "AimbotKey", 
    RageEnabled         = "RageKey", 
    HitNotifierEnabled  = "HitNotifierKeybind",
    RagebotEnabled      = "RagebotKey", 
    AntiAimEnabled      = "AntiAimKey", 
    CustomAnimEnabled   = "CustomAnimKey",
    ESPEnabled          = "ESPKey"
}

local function _getBoundKeyName(flag)
    local kpName = _keyMap[flag]
    if not kpName then return "" end
    local kp = Options[kpName] or Toggles[kpName]
    if not kp then return "" end
    local val = kp.Value
    if type(val) == "table" then val = val.Value end
    if not val or val == "None" or val == "" then return "" end
    return tostring(val)
end

local function _getMenuBindName()
    local kp = Options.MenuKeybind
    if not kp then return "None" end
    local val = kp.Value
    if type(val) == "table" then val = val.Value end
    if not val or val == "" then return "None" end
    return tostring(val)
end

local function _buildKeybindWindow()
    if _keybindWindow then return _keybindWindow end
    local sg = Instance.new("ScreenGui")
    sg.Name = "AXVV_KeybindWindow"; sg.ResetOnSpawn = false; sg.IgnoreGuiInset = true; sg.DisplayOrder = 999
    pcall(function() sg.Parent = ac end)
    if not sg.Parent then sg.Parent = LP:WaitForChild("PlayerGui") end

    local main = Instance.new("Frame")
    main.Name = "Main"; main.Size = UDim2.new(0, 240, 0, 80); main.Position = UDim2.new(0, 20, 0, 100)
    main.BackgroundColor3 = Color3.fromRGB(20, 20, 20); main.BackgroundTransparency = 0.1; main.BorderSizePixel = 0
    main.Parent = sg

    local stroke = Instance.new("UIStroke"); stroke.Color = Color3.fromRGB(0, 200, 255); stroke.Thickness = 1.5; stroke.Parent = main
    local corner = Instance.new("UICorner"); corner.CornerRadius = UDim.new(0, 6); corner.Parent = main

    local title = Instance.new("TextLabel")
    title.Name = "Title"; title.Size = UDim2.new(1, 0, 0, 26); title.BackgroundTransparency = 1
    title.Text = "Keybind"; title.TextColor3 = Color3.fromRGB(0, 200, 255)
    title.Font = Enum.Font.GothamBold; title.TextSize = 15; title.TextXAlignment = Enum.TextXAlignment.Center
    title.Parent = main

    local divider = Instance.new("Frame")
    divider.Size = UDim2.new(1, 0, 0, 1); divider.Position = UDim2.new(0, 0, 0, 26)
    divider.BackgroundColor3 = Color3.fromRGB(0, 200, 255); divider.BackgroundTransparency = 0.5
    divider.BorderSizePixel = 0; divider.Parent = main

    local list = Instance.new("Frame")
    list.Name = "List"; list.Size = UDim2.new(1, -12, 0, 0); list.Position = UDim2.new(0, 6, 0, 32)
    list.BackgroundTransparency = 1; list.Parent = main

    local layout = Instance.new("UIListLayout")
    layout.SortOrder = Enum.SortOrder.LayoutOrder; layout.Padding = UDim.new(0, 2); layout.Parent = list

    local dragState = { dragging = false, startMouse = Vector2.new(0,0), startPos = UDim2.new(0,0,0,0) }
    title.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragState.dragging = true
            dragState.startMouse = UserInputService:GetMouseLocation()
            dragState.startPos = main.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragState.dragging = false end
    end)
    RunService.Heartbeat:Connect(function()
        if not dragState.dragging or not main.Parent then return end
        local cur = UserInputService:GetMouseLocation()
        local delta = cur - dragState.startMouse
        main.Position = UDim2.new(
            dragState.startPos.X.Scale, dragState.startPos.X.Offset + delta.X,
            dragState.startPos.Y.Scale, dragState.startPos.Y.Offset + delta.Y
        )
    end)

    _keybindWindow = { gui = sg, main = main, list = list, layout = layout, title = title }
    return _keybindWindow
end

local function _destroyKeybindWindow()
    if _keybindWindow and _keybindWindow.gui then pcall(function() _keybindWindow.gui:Destroy() end) end
    _keybindWindow = nil; _keybindRows = {}
end

local function _refreshKeybindWindow()
    if not _keybindWindow then return end
    local win = _keybindWindow
    for _, lbl in ipairs(_keybindRows) do pcall(function() lbl:Destroy() end) end
    _keybindRows = {}

    -- Menu Keybind Row
    local menuRow = Instance.new("TextLabel")
    menuRow.Size = UDim2.new(1, 0, 0, 18); menuRow.BackgroundTransparency = 1
    menuRow.Font = Enum.Font.Gotham; menuRow.TextSize = 13
    menuRow.TextXAlignment = Enum.TextXAlignment.Left; menuRow.TextColor3 = Color3.fromRGB(0, 200, 255)
    menuRow.Text = "  Menu  [" .. _getMenuBindName() .. "]"
    menuRow.Parent = win.list; table.insert(_keybindRows, menuRow)

    -- Active Toggles
    local any = false
    for _, entry in ipairs(_keybindRegistry) do
        if _optIsOn(entry.flag) then
            any = true
            local row = Instance.new("TextLabel")
            row.Size = UDim2.new(1, 0, 0, 18); row.BackgroundTransparency = 1
            row.Font = Enum.Font.Gotham; row.TextSize = 13
            row.TextXAlignment = Enum.TextXAlignment.Left; row.TextColor3 = Color3.fromRGB(230, 230, 230)
            local key = _getBoundKeyName(entry.flag)
            row.Text = "  " .. entry.name .. (key ~= "" and ("  [" .. key .. "]") or "")
            row.Parent = win.list; table.insert(_keybindRows, row)
        end
    end

    if not any then
        local empty = Instance.new("TextLabel")
        empty.Size = UDim2.new(1, 0, 0, 18); empty.BackgroundTransparency = 1
        empty.Font = Enum.Font.Gotham; empty.TextSize = 13
        empty.TextColor3 = Color3.fromRGB(150, 150, 150); empty.TextXAlignment = Enum.TextXAlignment.Left
        empty.Text = "  (no active toggles)"; empty.Parent = win.list
        table.insert(_keybindRows, empty)
    end

    local count = 0
    for _, c in ipairs(win.list:GetChildren()) do if c:IsA("TextLabel") then count = count + 1 end end
    local height = 26 + (count * 20) + 8
    win.main.Size = UDim2.new(0, 240, 0, math.max(54, height))
end

task.spawn(function()
    while true do
        if _keybindWindow then pcall(_refreshKeybindWindow) end
        pcall(function() RunService.Heartbeat:Wait() end)
    end
end)

settingsMenu:AddToggle('KeybindMenu', {
    Text = 'Keybind Menu', Default = false,
    Callback = wrapToggleCB(function(Value)
        if Value then _buildKeybindWindow(); pcall(_refreshKeybindWindow) else _destroyKeybindWindow() end
    end),
})

        Text = 'Keybind Transparency', Default = 0, Min = 0, Max = 100, Rounding = 0, Suffix = '%',
        Callback = function(Value)
            pcall(_runAntiCheatBypass)
            local Alpha = Value / 100
            local Outer = bc.KeybindFrame
            if Outer then Outer.BackgroundTransparency = Alpha; Outer.BorderSizePixel = Alpha >= 1 and 0 or 1 end
            local Inner = bc.KeybindInner or (Outer and Outer:FindFirstChildOfClass('Frame'))
            if Inner then
                Inner.BackgroundTransparency = Alpha; Inner.BorderSizePixel = Alpha >= 1 and 0 or 1
                pcall(function() Inner:FindFirstChildOfClass('Frame').BackgroundTransparency = Alpha end)
            end
        end,
    })
end
