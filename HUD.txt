-- cache avatars
local thumbCache = {}

local function get_thumb(player)
    if thumbCache[player.UserId] then return thumbCache[player.UserId] end
    local ok, thumb = pcall(Players.GetUserThumbnailAsync, Players,
        player.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size150x150)
    if ok and thumb then
        thumbCache[player.UserId] = thumb
        return thumb
    end
    return nil
end

function HUD.set_target(player)
    if not player or not player.Parent then
        HUD.clear()
        return
    end
    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then
        HUD.clear()
        return
    end

    card.Visible = true
    dot.Visible = true

    local thumb = get_thumb(player)
    if thumb then avatar.Image = thumb end

    nameLabel.Text = player.DisplayName .. " (@" .. player.Name .. ")"

    local hp = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
    hpFill.Size = UDim2.new(hp, 0, 1, 0)

    if hp > 0.6 then
        hpFill.BackgroundColor3 = Color3.fromRGB(60, 220, 120)
    elseif hp > 0.3 then
        hpFill.BackgroundColor3 = Color3.fromRGB(230, 200, 60)
    else
        hpFill.BackgroundColor3 = Color3.fromRGB(230, 70, 70)
    end

    hpText.Text = string.format("%d / %d  HP", math.floor(hum.Health), math.floor(hum.MaxHealth))
end

function HUD.clear()
    card.Visible = false
    dot.Visible = false
end

function HUD.set_fov_visible(v)
    fovCircle.Visible = v and true or false
end

function HUD.set_fov_size(px)
    FOV_SIZE = px
    fovCircle.Size = UDim2.new(0, px, 0, px)
    fovCircle.Position = UDim2.new(0.5, -px/2, 0.5, -px/2)
end

function HUD.destroy()
    gui:Destroy()
end

return HUD
