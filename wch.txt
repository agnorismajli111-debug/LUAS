local function getWeaponCategory(name)
	for cat, list in pairs(WEAPON_CATEGORIES) do
		for _, n in ipairs(list) do
			if n == name then return cat end
		end
	end
	return nil
end

-- === Update text ===
local function updateHUD(weaponName)
	if weaponName == nil then
		textLabel.Text = "Weapons: (Idle) {Idle}"
		return
	end
	local category = getWeaponCategory(weaponName) or "Unknown"
	textLabel.Text = string.format("Weapons: (%s) {%s}", category, weaponName)
end

-- === Watch for YOUR equipped tool ===
local function hookCharacter(character)
	task.spawn(function()
		task.wait(0.2)
		for _, c in ipairs(character:GetChildren()) do
			if c:IsA("Tool") then
				updateHUD(c.Name)
				return
			end
		end
		updateHUD(nil)
	end)

	character.ChildAdded:Connect(function(child)
		if child:IsA("Tool") then
			updateHUD(child.Name)
		end
	end)

	character.ChildRemoved:Connect(function(child)
		if child:IsA("Tool") then
			task.wait(0.05)
			for _, c in ipairs(character:GetChildren()) do
				if c:IsA("Tool") then
					updateHUD(c.Name)
					return
				end
			end
			updateHUD(nil)
		end
	end)
end

if player.Character then hookCharacter(player.Character) end
player.CharacterAdded:Connect(hookCharacter)
